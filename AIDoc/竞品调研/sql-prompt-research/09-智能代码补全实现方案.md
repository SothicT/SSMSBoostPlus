# 09 - 智能代码补全：详细技术实现方案

> **目标**：构建一个可与 SSMS 深度集成的 T-SQL 智能代码补全引擎，达到 SQL Prompt 7-8 成功力。

### 数据流

```
用户键入字符 → 编辑器事件
  → 防抖 300ms
  → 提取当前语句文本
  → T-SQL 容错解析 → AST + 光标上下文
  → 上下文分类（FROM后/SELECT后/WHERE后/...）
  → 查询元数据缓存（按上下文类型过滤）
  → 候选排序（LRU权重 + 类型匹配 + 前缀匹配）
  → 填充 CompletionSet → SSMS 渲染候选列表
```

## 二、T-SQL 解析器选型与实现

### 2.1 三种方案对比

| 方案 | 优点 | 缺点 | 推荐指数 |
|------|------|------|---------|
| **ANTLR4 + TSQL Grammar** | 成熟语法文件、AST 完整、社区活跃 | JAR 依赖、性能中等、体积大 | ⭐⭐⭐⭐ |
| **手写递归下降解析器** | 零依赖、性能最优、可控性强 | 开发量大、容错需自实现 | ⭐⭐⭐⭐⭐ |
| **正则 + 状态机** | 实现最简单、体积最小 | 覆盖不全、维护困难、不支持复杂 SQL | ⭐⭐ |

### 2.2 推荐方案：混合架构

```
┌──────────────────────────────────────────────────────┐
│              混合解析策略                               │
├──────────────────────────────────────────────────────┤
│                                                      │
│  简单 SQL（单表查询、简单JOIN）                          │
│  ──────────────────────────                          │
│  使用手写状态机 + 正则快速解析                            │
│  覆盖 80% 日常场景，解析 < 1ms                          │
│                                                      │
│  复杂 SQL（CTE、子查询、UNION、窗口函数）                  │
│  ────────────────────────────────────────             │
│  使用 ANTLR4 完整解析，生成 AST                         │
│  解析 5-20ms，仅在快速路径失败时触发                     │
│                                                      │
│  容错处理（语法不完整时）                                │
│  ────────────────────────                             │
│  两种解析器都需支持错误恢复                              │
│  至少能识别到关键词位置级别                              │
│                                                      │
└──────────────────────────────────────────────────────┘
```

### 2.3 手写解析器核心实现

```csharp
/// <summary>
/// 轻量级 T-SQL 语法上下文识别器
/// 不构建完整 AST，只识别"光标在什么子句里"
/// </summary>
public class LightweightSqlParser
{
    // 关键字到上下文类型的映射表
    private static readonly Dictionary<string, SqlContext> KeywordContextMap = new(StringComparer.OrdinalIgnoreCase)
    {
        ["SELECT"]  = SqlContext.SelectList,
        ["FROM"]    = SqlContext.FromClause,
        ["WHERE"]   = SqlContext.WhereClause,
        ["JOIN"]    = SqlContext.JoinClause,
        ["INNER"]   = SqlContext.JoinClause,
        ["LEFT"]    = SqlContext.JoinClause,
        ["RIGHT"]   = SqlContext.JoinClause,
        ["ON"]      = SqlContext.OnCondition,
        ["ORDER"]   = SqlContext.OrderByClause,
        ["GROUP"]   = SqlContext.GroupByClause,
        ["HAVING"]  = SqlContext.HavingClause,
        ["INSERT"]  = SqlContext.InsertStatement,
        ["UPDATE"]  = SqlContext.UpdateStatement,
        ["DELETE"]  = SqlContext.DeleteStatement,
        ["SET"]     = SqlContext.SetClause,
        ["INTO"]    = SqlContext.IntoClause,
        ["VALUES"]  = SqlContext.ValuesClause,
        ["EXEC"]    = SqlContext.ExecuteStatement,
        ["ALTER"]   = SqlContext.AlterStatement,
        ["CREATE"]  = SqlContext.CreateStatement,
    };

    /// <summary>
    /// 解析光标在 SQL 文本中的上下文
    /// </summary>
    public ParseResult Parse(string sql, int cursorPosition)
    {
        // 步骤1：标记化 (Tokenization)
        var tokens = Tokenize(sql);
        
        // 步骤2：找到光标所在的 Token
        var cursorTokenIndex = FindCursorToken(tokens, cursorPosition);
        
        // 步骤3：回溯找到最近的关键字，确定上下文
        var context = DetermineContext(tokens, cursorTokenIndex);
        
        // 步骤4：提取已输入的过滤文本
        var filterText = ExtractFilterText(sql, cursorPosition, tokens, cursorTokenIndex);
        
        // 步骤5：检测表别名（用于列名补全时确定目标表）
        var tableAliases = DetectTableAliases(tokens, cursorTokenIndex);
        
        return new ParseResult(context, filterText, tableAliases);
    }

    private List<Token> Tokenize(string sql)
    {
        var tokens = new List<Token>();
        var current = new StringBuilder();
        var position = 0;
        var start = 0;
        var inString = false;
        var inComment = false;
        var inBracket = false;

        for (int i = 0; i < sql.Length; i++)
        {
            var ch = sql[i];

            // 处理字符串字面量
            if (ch == '\'' && !inComment && !inBracket)
            {
                inString = !inString;
                if (!inString)
                {
                    tokens.Add(new Token(TokenType.StringLiteral, current.ToString(), start, i + 1));
                    current.Clear();
                    start = i + 1;
                }
                continue;
            }
            if (inString) { current.Append(ch); continue; }

            // 处理方括号标识符
            if (ch == '[') { inBracket = true; current.Append(ch); continue; }
            if (ch == ']' && inBracket)
            {
                inBracket = false;
                current.Append(ch);
                tokens.Add(new Token(TokenType.Identifier, current.ToString(), start, i + 1));
                current.Clear();
                start = i + 1;
                continue;
            }
            if (inBracket) { current.Append(ch); continue; }

            // 处理注释 (-- 单行)
            if (ch == '-' && i + 1 < sql.Length && sql[i + 1] == '-' && !inComment)
            {
                // 吃掉当前 buffer
                FlushCurrent();
                inComment = true;
                continue;
            }
            if (inComment)
            {
                if (ch == '\n' || ch == '\r') inComment = false;
                continue;
            }

            // 空白字符：分割 token
            if (char.IsWhiteSpace(ch))
            {
                FlushCurrent();
                tokens.Add(new Token(TokenType.Whitespace, " ", i, i + 1));
                start = i + 1;
                continue;
            }

            // 运算符和分隔符
            if (IsOperatorOrDelimiter(ch))
            {
                FlushCurrent();
                tokens.Add(new Token(TokenType.Operator, ch.ToString(), i, i + 1));
                start = i + 1;
                continue;
            }

            current.Append(ch);
        }
        FlushCurrent();

        return tokens;

        void FlushCurrent()
        {
            if (current.Length == 0) return;
            var text = current.ToString();
            var type = KeywordContextMap.ContainsKey(text) 
                ? TokenType.Keyword 
                : TokenType.Identifier;
            tokens.Add(new Token(type, text, start, sql.Length));
            current.Clear();
        }
    }

    /// <summary>
    /// 回溯找到最近的关键词，确定上下文
    /// </summary>
    private SqlContext DetermineContext(List<Token> tokens, int cursorIndex)
    {
        SqlContext context = SqlContext.Unknown;
        string lastKeyword = null;
        
        // 从光标位置向后扫描，找到最近的关键词 + 上下文规则
        for (int i = cursorIndex - 1; i >= 0; i--)
        {
            if (tokens[i].Type != TokenType.Keyword) continue;
            
            var kw = tokens[i].Text.ToUpperInvariant();
            lastKeyword = kw;
            
            switch (kw)
            {
                case "FROM":
                case "JOIN":
                case "INNER":
                case "LEFT":
                case "RIGHT":
                case "FULL":
                case "CROSS":
                    // 在 FROM/JOIN 之后且下一个关键词之前 → 需要表名
                    if (context == SqlContext.Unknown)
                        context = SqlContext.FromClause;
                    break;
                    
                case "SELECT":
                    // 在 SELECT 之后 → 需要列名（但需要再往后看）
                    context = context == SqlContext.Unknown ? SqlContext.SelectList : context;
                    break;
                    
                case "ON":
                    context = context == SqlContext.Unknown ? SqlContext.OnCondition : context;
                    break;
                    
                case "WHERE":
                case "AND":
                case "OR":
                    context = context == SqlContext.Unknown ? SqlContext.WhereClause : context;
                    break;
                    
                case "ORDER":
                    context = context == SqlContext.Unknown ? SqlContext.OrderByClause : context;
                    break;
                    
                case "GROUP":
                    context = context == SqlContext.Unknown ? SqlContext.GroupByClause : context;
                    break;
                    
                case "INSERT":
                    context = SqlContext.InsertStatement;
                    break;
                    
                case "EXEC":
                case "EXECUTE":
                    context = SqlContext.ExecuteStatement;
                    break;
            }
        }
        
        return context;
    }
    
    private bool IsOperatorOrDelimiter(char ch)
    {
        return ch switch
        {
            ',' or '.' or '(' or ')' or '=' or '>' or '<' or '!' 
                or '+' or '-' or '*' or '/' or ';' => true,
            _ => false
        };
    }
}

// 数据类型定义
public enum SqlContext
{
    Unknown,
    SelectList,
    FromClause,
    WhereClause,
    JoinClause,
    OnCondition,
    OrderByClause,
    GroupByClause,
    HavingClause,
    InsertStatement,
    IntoClause,
    ValuesClause,
    UpdateStatement,
    SetClause,
    DeleteStatement,
    ExecuteStatement,
    AlterStatement,
    CreateStatement,
}

public enum TokenType { Keyword, Identifier, StringLiteral, Operator, Whitespace }

public record Token(TokenType Type, string Text, int StartIndex, int EndIndex);

public record ParseResult(
    SqlContext Context, 
    string FilterText, 
    Dictionary<string, string> TableAliases);
```

### 2.4 容错处理策略

```csharp
/// <summary>
/// 容错：即使是残缺的 SQL 也要给出合理上下文
/// </summary>
public SqlContext ParseWithFallback(string sql, int cursorPosition)
{
    try
    {
        // 主路径：手写解析器
        return LightweightParse(sql, cursorPosition).Context;
    }
    catch
    {
        // Fallback 1：纯正则匹配
        return RegexFallback(sql, cursorPosition);
    }
}

private SqlContext RegexFallback(string sql, int cursorPosition)
{
    // 简单粗暴：从光标位置往前找最后一个关键词
    var beforeCursor = sql[..cursorPosition];
    
    // 处理可能跨行的关键词
    var tokens = Regex.Matches(beforeCursor, @"\b(SELECT|FROM|WHERE|JOIN|ON|SET|INTO|EXEC)\b",
        RegexOptions.IgnoreCase | RegexOptions.Singleline);
    
    if (tokens.Count == 0) return SqlContext.Unknown;
    
    return tokens[^1].Value.ToUpperInvariant() switch
    {
        "SELECT" => SqlContext.SelectList,
        "FROM"   => SqlContext.FromClause,
        "JOIN"   => SqlContext.FromClause,
        "ON"     => SqlContext.OnCondition,
        "WHERE"  => SqlContext.WhereClause,
        "SET"    => SqlContext.SetClause,
        "INTO"   => SqlContext.IntoClause,
        "EXEC"   => SqlContext.ExecuteStatement,
        _        => SqlContext.Unknown
    };
}
```

### 3.2 缓存加载实现

```csharp
/// <summary>
/// Schema 缓存管理器
/// </summary>
public class SchemaCacheManager
{
    private readonly ConcurrentDictionary<string, DatabaseSchema> _caches = new();
    private readonly HashSet<string> _excludedDatabases = new();
    private readonly IProgress<CacheProgress> _progress;
    
    /// <summary>
    /// 异步加载数据库 Schema（不阻塞 SSMS UI）
    /// </summary>
    public async Task<DatabaseSchema> LoadOrGetAsync(
        string connectionString, 
        string databaseName,
        CancellationToken ct = default)
    {
        // 缓存命中
        if (_caches.TryGetValue(databaseName, out var cached))
            return cached;
        
        // 排除列表命中
        if (IsExcluded(databaseName))
            return null;
        
        var schema = new DatabaseSchema { DatabaseName = databaseName };
        
        try
        {
            using var conn = new SqlConnection(connectionString);
            await conn.OpenAsync(ct);
            
            _progress?.Report(new CacheProgress(databaseName, CachePhase.Loading, 0));
            
            // 并行加载各类对象
            var loadTasks = new[]
            {
                LoadTablesAsync(conn, schema, ct),
                LoadViewsAsync(conn, schema, ct),
                LoadProceduresAsync(conn, schema, ct),
                LoadFunctionsAsync(conn, schema, ct),
            };
            
            await Task.WhenAll(loadTasks);
            
            schema.LoadedAt = DateTime.UtcNow;
            _caches[databaseName] = schema;
            
            _progress?.Report(new CacheProgress(databaseName, CachePhase.Completed, 100));
            
            return schema;
        }
        catch (Exception ex)
        {
            _progress?.Report(new CacheProgress(databaseName, CachePhase.Failed, 0, ex.Message));
            return null;
        }
    }
    
    /// <summary>
    /// 加载表 + 列 + 外键（最核心的数据）
    /// </summary>
    private async Task LoadTablesAsync(
        SqlConnection conn, DatabaseSchema schema, CancellationToken ct)
    {
        // 使用 INFORMATION_SCHEMA + sys 视图的组合查询，一次获取表/列/键信息
        const string sql = @"
SELECT 
    s.name AS SchemaName,
    t.name AS TableName,
    c.name AS ColumnName,
    ty.name AS DataType,
    c.max_length,
    c.precision,
    c.scale,
    c.is_nullable,
    c.column_id,
    CASE WHEN pk.column_id IS NOT NULL THEN 1 ELSE 0 END AS IsPrimaryKey,
    CASE WHEN fk.parent_column_id IS NOT NULL THEN 1 ELSE 0 END AS IsForeignKey,
    CASE WHEN c.is_identity = 1 THEN 1 ELSE 0 END AS IsIdentity,
    t.is_ms_shipped AS IsSystemTable
FROM sys.tables t
INNER JOIN sys.schemas s ON t.schema_id = s.schema_id
INNER JOIN sys.columns c ON t.object_id = c.object_id
INNER JOIN sys.types ty ON c.user_type_id = ty.user_type_id
LEFT JOIN (
    SELECT ic.object_id, ic.column_id 
    FROM sys.index_columns ic
    INNER JOIN sys.indexes i ON ic.object_id = i.object_id AND ic.index_id = i.index_id
    WHERE i.is_primary_key = 1
) pk ON c.object_id = pk.object_id AND c.column_id = pk.column_id
LEFT JOIN sys.foreign_key_columns fk 
    ON c.object_id = fk.parent_object_id AND c.column_id = fk.parent_column_id
ORDER BY s.name, t.name, c.column_id";
        
        using var cmd = new SqlCommand(sql, conn);
        cmd.CommandTimeout = 30; // 大型数据库需要较长超时
        
        using var reader = await cmd.ExecuteReaderAsync(ct);
        
        var tables = new Dictionary<string, TableInfo>(StringComparer.OrdinalIgnoreCase);
        
        while (await reader.ReadAsync(ct))
        {
            var schemaName = reader.GetString(0);
            var tableName = reader.GetString(1);
            var fullName = $"[{schemaName}].[{tableName}]";
            
            if (!tables.TryGetValue(fullName, out var table))
            {
                table = new TableInfo
                {
                    Schema = schemaName,
                    Name = tableName,
                    IsSystemTable = reader.GetBoolean(12)
                };
                tables[fullName] = table;
            }
            
            // 跳过系统表（可通过配置控制）
            if (table.IsSystemTable)
                continue;
            
            table.Columns.Add(new ColumnInfo
            {
                Name = reader.GetString(2),
                DataType = GetFullDataType(reader),
                BaseDataType = reader.GetString(3),
                IsPrimaryKey = reader.GetBoolean(8),
                IsForeignKey = reader.GetBoolean(9),
                IsIdentity = reader.GetBoolean(10),
                IsNullable = reader.GetBoolean(6),
                OrdinalPosition = reader.GetInt32(7)
            });
        }
        
        schema.Tables = tables;
    }
    
    private static string GetFullDataType(SqlDataReader reader)
    {
        var type = reader.GetString(3);
        var maxLen = reader.GetInt16(4);
        
        return type.ToLower() switch
        {
            "nvarchar" or "varchar" or "nchar" or "char" 
                => maxLen == -1 ? $"{type}(max)" : $"{type}({maxLen})",
            "decimal" or "numeric" 
                => $"{type}({reader.GetByte(5)},{reader.GetByte(6)})",
            _ => type
        };
    }
}
```

```csharp
public class CacheRefreshPolicy
{
    /// <summary>
    /// 手动刷新（替代 SQL Prompt 的 Ctrl+Shift+D）
    /// 仅刷新当前数据库

    }
    
    /// <summary>
    /// 自动刷新（实验性）
    /// 监听 DDL 变更后自动刷新缓存
    /// </summary>
    public async Task EnableAutoRefresh(string databaseName, string connectionString)
    {
        // 方案1：使用 SQL Server Service Broker 监听 DDL 事件
        // 方案2：定期轮询 sys.objects 的 modify_date
        // 方案3：解析用户执行的 DDL 语句执行后立即刷新
        
        // 推荐方案3：在 execute 事件中检查是否包含 DDL，如果有则延迟刷新
        // 无需额外查询，性能最优
    }
    
    /// <summary>
    /// 内存管理：排除特定对象
    /// </summary>

        // 支持通配符：*表示匹配任意字符，?表示匹配单个字符

        //   "tempdb"    → 排除整个 tempdb
```

## 四、补全引擎核心实现

### 4.1 上下文 → 候选映射

```csharp
/// <summary>
/// 根据解析出的上下文，查询元数据缓存生成候选列表
/// </summary>
public class CompletionEngine
{
    private readonly SchemaCacheManager _cache;
    private readonly UsageTracker _usage;
    
    /// <summary>
    /// 生成补全候选列表
    /// </summary>
    public List<CompletionCandidate> Generate(
        ParseResult parseResult, 
        string databaseName, 
        Dictionary<string, string> tableAliases)
    {
        var candidates = new List<CompletionCandidate>();
        var schema = _cache.Get(databaseName);
        if (schema == null) return candidates;
        
        var ctx = parseResult.Context;
        var filter = parseResult.FilterText;
        
        switch (ctx)
        {
            case SqlContext.SelectList:
                // 在 SELECT 后：需要列名
                candidates.AddRange(GetColumnCandidates(schema, tableAliases, filter));
                candidates.AddRange(GetFunctionCandidates(schema, filter));
                // 也显示关键词（FROM, INTO 等）
                candidates.AddRange(GetKeywordCandidates(ctx, filter));
                break;
                
            case SqlContext.FromClause:
            case SqlContext.JoinClause:
                // 在 FROM/JOIN 后：需要表名/视图名
                candidates.AddRange(GetTableCandidates(schema, filter));
                candidates.AddRange(GetViewCandidates(schema, filter));
                break;
                
            case SqlContext.OnCondition:
                // JOIN ON 条件：需要关联表的列名
                candidates.AddRange(GetColumnCandidates(schema, tableAliases, filter));
                break;
                
            case SqlContext.WhereClause:
                // WHERE 条件：需要列名 + 函数
                candidates.AddRange(GetColumnCandidates(schema, tableAliases, filter));
                candidates.AddRange(GetFunctionCandidates(schema, filter));
                break;
                
            case SqlContext.ExecuteStatement:
                // EXEC 后：需要存储过程名
                candidates.AddRange(GetProcedureCandidates(schema, filter));
                break;
                
            case SqlContext.InsertStatement:
            case SqlContext.IntoClause:
                candidates.AddRange(GetTableCandidates(schema, filter));
                break;
                
            case SqlContext.OrderByClause:
            case SqlContext.GroupByClause:
                candidates.AddRange(GetColumnCandidates(schema, tableAliases, filter));
                break;
                
            default:
                candidates.AddRange(GetKeywordCandidates(ctx, filter));
                break;
        }
        
        // 应用统一排序
        return Ranker.Sort(candidates, ctx, _usage.GetHistory());
    }
    
    private IEnumerable<CompletionCandidate> GetTableCandidates(DatabaseSchema schema, string filter)
    {
        return schema.Tables.Values
            .Where(t => !t.IsSystemTable)
            .Where(t => MatchesFilter(t.FullName, filter))
            .Select(t => new CompletionCandidate
            {
                DisplayText = t.Name,
                InsertionText = t.FullName,
                Description = $"表 ({t.Columns.Count} 列)",
                Category = CompletionCategory.Table,
                SortPriority = 10,
                Glyph = GlyphType.Table
            });
    }
    
    private IEnumerable<CompletionCandidate> GetColumnCandidates(
        DatabaseSchema schema, Dictionary<string, string> aliases, string filter)
    {
        var columns = new List<CompletionCandidate>();
        
        if (aliases != null && aliases.Any())
        {
            // 有表别名：只提示该表的列（更精准）
            foreach (var (alias, tableName) in aliases)
            {
                if (schema.Tables.TryGetValue(tableName, out var table))
                {
                    columns.AddRange(table.Columns
                        .Where(c => MatchesFilter(c.Name, filter))
                        .Select(c => new CompletionCandidate
                        {
                            DisplayText = $"{alias}.{c.Name}",
                            InsertionText = $"[{alias}].[{c.Name}]",
                            Description = $"{table.Name}.{c.Name} ({c.DataType})",
                            Category = CompletionCategory.Column,
                            SortPriority = 20,
                            Glyph = c.IsPrimaryKey ? GlyphType.PrimaryKey : GlyphType.Column
                        }));
                }
            }
        }
        else
        {
            // 无别名：列出所有表的列
            foreach (var table in schema.Tables.Values.Where(t => !t.IsSystemTable))
            {
                columns.AddRange(table.Columns
                    .Where(c => MatchesFilter(c.Name, filter))
                    .Select(c => new CompletionCandidate
                    {
                        DisplayText = c.Name,
                        InsertionText = $"[{c.Name}]",
                        Description = $"{table.Name}.{c.Name} ({c.DataType})",
                        Category = CompletionCategory.Column,
                        SortPriority = 15,
                        Glyph = c.IsPrimaryKey ? GlyphType.PrimaryKey : GlyphType.Column
                    }));
            }
        }
        
        return columns;
    }
}
```

### 4.2 排序算法实现

```csharp
/// <summary>
/// 多因子加权排序算法
/// </summary>
public static class Ranker
{
    public static List<CompletionCandidate> Sort(
        List<CompletionCandidate> candidates,
        SqlContext context,
        IReadOnlyDictionary<string, int> usageHistory)
    {
        foreach (var candidate in candidates)
        {
            float score = 0;
            
            // === 因子1：使用频率 (权重 40%) ===
            if (usageHistory.TryGetValue(candidate.DisplayText, out var useCount))
            {
                // 对数归一化，避免极高频项统治列表
                score += (float)Math.Log(useCount + 1) * 100;
            }
            
            // === 因子2：最近使用 (权重 25%) ===
            if (usageHistory.TryGetValue(candidate.DisplayText, out var recencyOrder))
            {
                // recencyOrder 越小越近，给高分
                score += 1000f / (recencyOrder + 1);
            }
            
            // === 因子3：上下文匹配 (权重 20%) ===
            score += (candidate.Category, context) switch
            {
                (CompletionCategory.Table, SqlContext.FromClause) => 200,
                (CompletionCategory.Column, SqlContext.SelectList) => 200,
                (CompletionCategory.Column, SqlContext.WhereClause) => 150,
                (CompletionCategory.Procedure, SqlContext.ExecuteStatement) => 200,
                (CompletionCategory.Keyword, _) => 50,
                _ => 0
            };
            
            // === 因子4：前缀匹配 (权重 15%) ===
            // (filter 匹配已在 Candidates 生成时过滤，这里微调)
            score += candidate.SortPriority * 2;
            
            candidate.RankScore = score;
        }
        
        return candidates
            .OrderByDescending(c => c.RankScore)
            .ThenBy(c => c.DisplayText, StringComparer.OrdinalIgnoreCase)
            .Take(50) // 限制候选数量，避免 UI 卡顿
            .ToList();
    }
}

public class CompletionCandidate
{
    public string DisplayText { get; set; }       // 列表显示文本
    public string InsertionText { get; set; }     // 选中后插入的文本
    public string Description { get; set; }       // 描述/工具提示
    public CompletionCategory Category { get; set; }
    public int SortPriority { get; set; }
    public GlyphType Glyph { get; set; }
    public float RankScore { get; set; }
}

public enum CompletionCategory { Table, View, Column, Procedure, Function, Keyword, Snippet }
public enum GlyphType { Table, View, Column, PrimaryKey, Procedure, Function, Keyword }
```

```csharp
/// <summary>
/// 基于 LRU 的使用历史跟踪器
/// 持久化到文件，跨 SSMS 会话保留
/// </summary>
public class UsageTracker
{
    private readonly Dictionary<string, UsageEntry> _store = new();
    private readonly string _persistPath;
    private int _sequenceNumber = 0;  // 自增序号，表示最近使用的顺序
    
    public UsageTracker(string persistPath)
    {
        _persistPath = persistPath;
        Load();
    }
    
    /// <summary>
    /// 记录一次使用
    /// </summary>
    public void RecordUse(string itemKey)
    {
        if (_store.TryGetValue(itemKey, out var entry))
        {
            entry.UseCount++;
            entry.LastUsed = DateTime.UtcNow;
            entry.RecencyRank = ++_sequenceNumber;
        }
        else
        {
            _store[itemKey] = new UsageEntry
            {
                Key = itemKey,
                UseCount = 1,
                FirstUsed = DateTime.UtcNow,
                LastUsed = DateTime.UtcNow,
                RecencyRank = ++_sequenceNumber
            };
        }
        
        // 异步定期持久化
        ScheduleSave();
    }
    
    public IReadOnlyDictionary<string, int> GetHistory()
    {
        // 返回 useCount 用于排序
        return _store.ToDictionary(kvp => kvp.Key, kvp => kvp.Value.UseCount);
    }
    
    public IReadOnlyDictionary<string, int> GetRecency()
    {
        // 返回 recencyRank 用于排序
        // 需要转换为倒序（最近使用的排前面）
        var maxRank = _sequenceNumber;
        return _store.ToDictionary(kvp => kvp.Key, kvp => maxRank - kvp.Value.RecencyRank);
    }
    
    private void Save()
    {
        var data = JsonSerializer.Serialize(_store.Values);
        File.WriteAllText(_persistPath, data);
    }
    
    private void Load()
    {
        if (!File.Exists(_persistPath)) return;
        try
        {
            var data = JsonSerializer.Deserialize<List<UsageEntry>>(
                File.ReadAllText(_persistPath));
            foreach (var entry in data)
            {
                _store[entry.Key] = entry;
                _sequenceNumber = Math.Max(_sequenceNumber, entry.RecencyRank);
            }
        }
        catch { /* 文件损坏则丢弃历史 */ }
    }
    
    private class UsageEntry
    {
        public string Key { get; set; }
        public int UseCount { get; set; }
        public DateTime FirstUsed { get; set; }
        public DateTime LastUsed { get; set; }
        public int RecencyRank { get; set; }
    }
}
```

```csharp
/// <summary>
/// 补全源：SSMS 编辑器在每次用户输入时调用 AugmentCompletionSession
/// </summary>
internal class SqlCompletionSource : ICompletionSource
{
    private readonly ITextBuffer _buffer;
    private readonly CompletionEngine _engine;
    private readonly SchemaCacheManager _cacheManager;
    private readonly LightweightSqlParser _parser = new();
    private DateTime _lastCompletionTime = DateTime.MinValue;
    
    public SqlCompletionSource(
        ITextBuffer buffer, 
        CompletionEngine engine, 
        SchemaCacheManager cacheManager)
    {
        _buffer = buffer;
        _engine = engine;
        _cacheManager = cacheManager;
    }
    
    /// <summary>
    /// SSMS 调用此方法来获取补全列表
    /// 用户每输入一个字符通常都会触发
    /// </summary>
    public void AugmentCompletionSession(
        ICompletionSession session, 
        IList<CompletionSet> completionSets)
    {
        // === 防抖：限制触发频率 ===
        var now = DateTime.UtcNow;
        if ((now - _lastCompletionTime).TotalMilliseconds < 200)
            return;
        _lastCompletionTime = now;
        
        // === 获取文本和光标位置 ===
        var snapshot = _buffer.CurrentSnapshot;
        var triggerPoint = session.GetTriggerPoint(_buffer).GetPoint(snapshot);
        var line = triggerPoint.GetContainingLine();
        
        // 只解析从语句头到光标位置的文本
        var statementStart = FindStatementStart(line);
        var textToCursor = snapshot.GetText(
            Span.FromBounds(statementStart, triggerPoint.Position));
        
        // === 解析 SQL 上下文 ===
        var parseResult = _parser.Parse(textToCursor, textToCursor.Length);
        
        // 如果没有明确上下文，不提供补全（让 SSMS 默认补全接管）
        if (parseResult.Context == SqlContext.Unknown) return;
        
        // === 获取数据库连接信息 ===
        var databaseName = GetCurrentDatabase(snapshot.TextBuffer);
        if (string.IsNullOrEmpty(databaseName)) return;
        
        // === 生成候选列表 ===
        var candidates = _engine.Generate(parseResult, databaseName, 
            parseResult.TableAliases);
        if (!candidates.Any()) return;
        
        // === 创建 CompletionSet ===
        var applicableTo = snapshot.CreateTrackingSpan(
            triggerPoint.Position - parseResult.FilterText.Length,
            parseResult.FilterText.Length,
            SpanTrackingMode.EdgeInclusive);
        
        var completions = candidates.Select(c => new Completion(
            displayText: c.DisplayText,
            insertionText: c.InsertionText,
            description: c.Description,
            iconSource: GetGlyphImage(c.Glyph),
            iconAutomationText: c.Category.ToString()
        ));
        
        var completionSet = new CompletionSet(
            moniker: "SqlIntelliSense",
            displayName: "SQL 智能补全",
            applicableTo: applicableTo,
            completions: completions,
            matchOptions: null  // 使用默认前缀匹配
        );
        
        // 设置过滤器（大小写不敏感的前缀匹配）
        completionSet.Filter(c => c.DisplayText, (text, pattern) =>
            text.StartsWith(pattern, StringComparison.OrdinalIgnoreCase), 
            StringComparer.OrdinalIgnoreCase);
        
        completionSets.Add(completionSet);
    }
    
    /// <summary>
    /// 找到当前语句的起始位置
    /// 支持多语句（用分号或 GO 分隔）
    /// </summary>
    private int FindStatementStart(ITextSnapshotLine line)
    {
        var lineNumber = line.LineNumber;
        var snapshot = line.Snapshot;
        
        for (int i = lineNumber; i >= 0; i--)
        {
            var currentLine = snapshot.GetLineFromLineNumber(i);
            var text = currentLine.GetText().Trim().ToUpperInvariant();
            
            // 前一行以分号结尾 → 这是语句起点
            // 前一行是 GO → 这是语句起点
            // 前面的语句已结束
            if (text.EndsWith(";") || text == "GO" || i == 0)
                return currentLine.Start.Position;
        }
        
        return 0;
    }
    
    /// <summary>
    /// 获取当前查询窗口的数据库名称
    /// 通过 SSMS 的 DTE 自动化模型实现
    /// </summary>
    private string GetCurrentDatabase(ITextBuffer buffer)
    {
        // 方式1：通过 ITextDocument 获取文件属性中的连接信息
        // 方式2：通过解析 SQL 文本中的 USE 语句
        // 方式3：通过 DTE2.ActiveDocument 获取连接属性
        
        // 简化实现：默认返回当前连接的默认数据库
        // 完整实现需要与 SSMS 的连接管理集成
        return "CurrentDatabase"; // 占位符
    }
    
    public void Dispose() { }
}
```

```csharp
/// <summary>
/// 延迟加载：先加载表名，展开表时才加载列
/// </summary>
public class LazySchemaCache
{
    private readonly Dictionary<string, List<string>> _tableNames = new();
    private readonly Dictionary<string, Task<TableDetail>> _pendingLoads = new();
    
    /// <summary>
    /// 快速加载所有表名（< 1s，即使是大数据库）
    /// </summary>
    public async Task LoadTableNamesAsync(SqlConnection conn)
    {
        const string sql = @"
SELECT s.name + '.' + t.name 
FROM sys.tables t 
INNER JOIN sys.schemas s ON t.schema_id = s.schema_id
WHERE t.is_ms_shipped = 0";
        
        using var cmd = new SqlCommand(sql, conn);
        using var reader = await cmd.ExecuteReaderAsync();
        
        while (await reader.ReadAsync())
        {
            _tableNames[reader.GetString(0)] = null; // 表名已加载，列待加载
        }
    }
    
    /// <summary>
    /// 按需加载列的详细信息
    /// </summary>
    public async Task<TableDetail> LoadTableDetailAsync(
        SqlConnection conn, string tableName)
    {
        if (_pendingLoads.TryGetValue(tableName, out var pending))
            return await pending;
        
        var task = LoadTableDetailInternal(conn, tableName);
        _pendingLoads[tableName] = task;
        return await task;
    }
}
```

## 九、总结

SQL Prompt 的智能代码补全是其核心功能，复刻该功能在技术上是完全可行的。关键成功要素：

1. **T-SQL 解析**：推荐混合架构（手写快速路径 + ANTLR 复杂路径），兼顾性能与覆盖率
2. **元数据缓存**：从 `sys.tables/columns` 一次性加载，内存中查询，响应 < 50ms
3. **编辑器集成**：通过 VS Shell `ICompletionSource` 实现，这是最简单直接的集成方式
4. **渐进式交付**：先做表名/列名补全 MVP（4 周），再逐步增加高级功能
